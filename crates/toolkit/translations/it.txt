# GolemSculpt tooltips in Italian (it).
#
# Each tooltip is a pair of lines: "en:" is the English the app has, "it:"
# is what the tooltip shows instead. To fix a translation, edit its
# "it:" line. Words in {braces} are filled in by the app and must stay;
# \n is a line break. Names of buttons and cards stay in English, as they
# are on screen. tools/translations.py adds what is new.
#
# The words used throughout:
#   mesh = mesh (la mesh)
#   model = modello
#   object = oggetto
#   scene = scena
#   surface = superficie
#   vertex / vertices = vertice / vertici
#   edge = spigolo
#   face = faccia
#   triangle = triangolo
#   quad = quad
#   brush = pennello
#   stroke = pennellata
#   dab = tocco
#   sculpt (verb) = scolpire
#   mask = maschera
#   symmetry = simmetria
#   mirror (verb) = specchiare
#   mirror image = immagine speculare
#   mirror plane = piano di simmetria / di specchio
#   subdivision level = livello di suddivisione
#   remesh = Remesh (resta com'è)
#   decimate = decimare (il pulsante resta Decimate)
#   falloff = decadimento
#   alpha = alpha
#   layer (sculpt, paint) = layer
#   face group = gruppo di facce
#   pivot = pivot
#   gizmo = gizmo
#   viewport = viewport
#   card = riquadro
#   panel = pannello
#   tab = scheda
#   undo = Undo (resta com'è, è il nome del pulsante)
#   strength = intensità (la Strength con l'iniziale maiuscola resta)
#   radius = raggio
#   size = dimensione
#   smooth (verb) = levigare
#   inflate = gonfiare
#   bend / twist = curvare / torcere
#   paint (noun) = pittura
#   paint (verb) = dipingere
#   colour = colore
#   picture = immagine
#   world's axes / own axes = assi globali / assi locali
#   joint (limb, sphere rig) = articolazione
#   joint (Keys tool) = giunto; peg = perno; socket = sede
#   recovery copy = copia di recupero
#   drag / click / right-click = trascinare / clic / clic destro
#   bake = bake (eseguire il bake)
