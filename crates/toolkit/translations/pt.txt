# GolemSculpt tooltips in Portuguese (Brazil) (pt).
#
# Each tooltip is a pair of lines: "en:" is the English the app has, "pt:"
# is what the tooltip shows instead. To fix a translation, edit its
# "pt:" line. Words in {braces} are filled in by the app and must stay;
# \n is a line break. Names of buttons and cards stay in English, as they
# are on screen. tools/translations.py adds what is new.
#
# The words used throughout:
#   mesh = malha
#   vertex / vertices = vértice / vértices
#   edge = aresta
#   face = face
#   triangle = triângulo
#   quad = quad
#   surface = superfície
#   model = modelo
#   object = objeto
#   scene = cena
#   brush = pincel
#   stroke = traço
#   dab = toque
#   mask / masked = máscara / mascarado
#   symmetry = simetria
#   mirror (verb) / mirrored = espelhar / espelhado
#   mirror plane = plano de espelhamento
#   subdivision level = nível de subdivisão
#   remesh (noun, verb) = remesh, refazer a malha
#   decimate (verb) = decimar
#   falloff = falloff
#   alpha = alpha
#   layer = camada
#   face group = grupo de faces
#   pivot = pivô
#   gizmo = gizmo
#   viewport = viewport
#   view = vista
#   card = card
#   panel = painel
#   tab = aba
#   Undo / Redo (the buttons) = Undo / Redo, kept in English
#   strength (generic) = força
#   radius = raio
#   size = tamanho
#   smooth (verb) = suavizar
#   inflate = inflar
#   crease = vinco
#   cage = gaiola
#   live (copy, array) = dinâmico(a)
#   picture = imagem
#   joint (rig, limb) = articulação
#   joint (Keys tool) / peg / socket = junta / pino / furo
#   link (Blender) / link (chain) = vínculo / elo
#   slider = controle deslizante
#   range = intervalo
#   paint = pintura
#   pen = caneta
#   turntable = turntable
