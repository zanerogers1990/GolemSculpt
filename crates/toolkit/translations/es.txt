# GolemSculpt tooltips in Spanish (es).
#
# Each tooltip is a pair of lines: "en:" is the English the app has, "es:"
# is what the tooltip shows instead. To fix a translation, edit its
# "es:" line. Words in {braces} are filled in by the app and must stay;
# \n is a line break. Names of buttons and cards stay in English, as they
# are on screen. tools/translations.py adds what is new.
#
# The words used throughout:
#   mesh = malla
#   vertex = vértice
#   edge = arista
#   face = cara
#   triangle = triángulo
#   surface = superficie
#   model = modelo
#   object = objeto
#   scene = escena
#   brush = pincel
#   brush in hand = pincel en uso
#   stroke = trazo
#   dab = toque
#   sculpt (verb) = esculpir
#   mask = máscara
#   masked = enmascarado
#   symmetry = simetría
#   mirror (noun) = espejo
#   mirror (verb) = reflejar
#   mirror image = imagen especular
#   subdivision level = nivel de subdivisión
#   remesh (verb) = remallar
#   decimate (verb) = decimar
#   falloff = caída
#   alpha = alfa
#   print (alpha stamp) = huella
#   layer = capa
#   paint layer = capa de pintura
#   face group = grupo de caras
#   pivot = pivote
#   gizmo = gizmo
#   handle = tirador
#   viewport = vista 3D
#   view = vista
#   card = tarjeta
#   panel = panel
#   tab = pestaña
#   slider = deslizador
#   undo = deshacer
#   strength = intensidad
#   radius = radio
#   smooth = suavizar
#   inflate = inflar
#   crease = pliegue
#   bake = hornear
#   cage = jaula
#   hidden / showing = oculto / visible
#   click / right-click = clic / clic derecho
#   press (key or button) = pulsar
#   pen = lápiz
